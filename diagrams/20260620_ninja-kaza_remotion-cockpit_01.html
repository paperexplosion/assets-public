    Kaza 自動収録スクリプトを作る（~/kaza-tts/record_kaza.py）
    """ Kaza 自動収録スクリプト - Chatterbox WAV を再生しながら - VTube Studio API で台本に沿って表情を自動切り替え - OBS は別途起動して録画しておく（または OBS WebSocket API で制御） 使い方: python record_kaza.py script.json """ import asyncio, json, subprocess, time, sys from pathlib import Path import websockets VTUBE_WS = "ws://localhost:8001" OUTPUT_DIR = Path("output") # 表情名 → VTube Studio の HotkeyID（事前に設定） EXPRESSION_MAP = { "neutral": "hotkey_neutral", "happy": "hotkey_happy", "surprised": "hotkey_surprised", "thinking": "hotkey_thinking", "serious": "hotkey_serious", "pose1": "hotkey_ninja_pose1", "pose2": "hotkey_ninja_pose2", } async def trigger_expression(ws, token, expr_name): hotkey_id = EXPRESSION_MAP.get(expr_name) if not hotkey_id: return req = { "apiName": "VTubeStudioPublicAPI", "apiVersion": "1.0", "messageType": "HotkeyTriggerRequest", "data": {"hotkeyID": hotkey_id} } await ws.send(json.dumps(req)) await ws.recv() async def record(script_path): with open(script_path) as f: segments = json.load(f) # 例: [{"id":"01","wav":"output/01.wav","expr":"neutral","duration":12.5}, ...] async with websockets.connect(VTUBE_WS) as ws: # 認証 auth_req = {"apiName":"VTubeStudioPublicAPI","apiVersion":"1.0", "messageType":"AuthenticationRequest", "data":{"pluginName":"KazaRecorder","pluginDeveloper":"Suzuki", "authenticationToken":""}} await ws.send(json.dumps(auth_req)) auth_res = json.loads(await ws.recv()) token = auth_res.get("data", {}).get("authenticationToken", "") for seg in segments: print(f"▶ {seg['id']}: {seg.get('expr','neutral')}") await trigger_expression(ws, token, seg.get("expr", "neutral")) # WAV 再生（BlackHole 経由で VTube Studio がリップシンク） proc = subprocess.Popen(["afplay", str(OUTPUT_DIR / f"{seg['id']}.wav")]) proc.wait() await asyncio.sleep(0.3) # 次セグメントまで間 if __name__ == "__main__": asyncio.run(record(sys.argv[1])) VTube Studio API websockets (Python) afplay (macOS 標準)

PHASE 4 ― Editor Starter セットアップ（✓ 入手済み）
✓ 入手済みRemotion Studio の次のステップとして即導入できる
Editor Starter 入手済み。 Remotion プロジェクト（Phase 1）が完成したら即導入できる。Claude Code でコードを書きつつ、タイムラインの視覚調整は Editor Starter の GUI を使う「両刀運用」が最強構成。
Editor Starter の得意な作業：
・タイムライン上でクリップの長さをドラッグ調整
・テキスト位置・フォント・カラーを目で見ながら微調整
・Asset upload からファイルをドラッグで素早く差し込む
・クライアントや自分でプレビューしながらリアルタイム修正

Claude Code の得意な作業（声で指示）：
・シーン・Sequence の追加・削除・並び替え
・タイミング（from / durationInFrames）の数値変更
・コンポーネントの構造変更・新 props 追加
・BGM・テキストオーバーレイの一括適用
